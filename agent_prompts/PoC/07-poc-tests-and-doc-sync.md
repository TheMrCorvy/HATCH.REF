```dart
      final result = computeElapsedOfflineState(

        currentTime: t2h,
        isActive: true,

      expect(result.energy, 72);

        hunger: 80,

        lastInteractionTimestamp: t0,

        currentTime: t10h,
        isActive: false,
```

```dart
void main() {
  group('generateKaijuName', () {

      for (var i = 0; i < 30; i++) {
        final name = generateKaijuName('godzilla', random: Random(i));

      expect(godzillaPool.contains(fallbackName), isTrue);
    });
```

```dart
import 'package:flutter_test/flutter_test.dart';
import 'package:tamagotchi_mobile/models/age_phase.dart';
import 'package:tamagotchi_mobile/models/kaiju_state.dart';
import 'package:tamagotchi_mobile/plugins/executors/generic_customization_renderer.dart';
import 'package:tamagotchi_mobile/plugins/executors/generic_user_action_executor.dart';
import 'package:tamagotchi_mobile/plugins/models/customization_plugin_config.dart';
import 'package:tamagotchi_mobile/plugins/plugin_registry.dart';

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();

  group('PluginRegistry & GenericUserActionExecutor', () {
    setUp(() async {
      await PluginRegistry().loadBundledPocManifest();
    });

    test('loads eat, sleep, play actions and 4 palettes from bundled manifest', () {
      final actions = PluginRegistry().getEnabledUserActions(
        kaijuPhase: AgePhase.child,
      );
      expect(actions.map((a) => a.id).toList(), ['eat', 'sleep', 'play']);

      final palettes = PluginRegistry()
          .getCustomizationsByCategory(CustomizationCategory.palette);
      expect(palettes.length, 4);
    });

    test('executes eat action with +25 hunger, -5 energy, +5 happiness', () {
      final eatConfig = PluginRegistry()
          .getEnabledUserActions(kaijuPhase: AgePhase.child)
          .firstWhere((a) => a.id == 'eat');
      final kaiju = KaijuInstance(
        id: 'k1',
        kaijuType: 'godzilla',
        nickname: 'CRAZY_DIAMOND_PULSE',
        lastInteractionTimestamp: DateTime(2026, 10, 31),
      );

      final result = GenericUserActionExecutor.execute(
        currentKaiju: kaiju,
        config: eatConfig,
        triggeredAt: DateTime(2026, 10, 31),
      );

      expect(result.success, isTrue);
      expect(result.hungerDelta, 25);
      expect(result.energyDelta, -5);
      expect(result.happinessDelta, 5);
      expect(result.animationSequence, 'eating');
    });

    test('rejects play action when Kaiju energy < 15', () {
      final playConfig = PluginRegistry()
          .getEnabledUserActions(kaijuPhase: AgePhase.child)
          .firstWhere((a) => a.id == 'play');
      final exhaustedKaiju = KaijuInstance(
        id: 'k1',
        kaijuType: 'godzilla',
        nickname: 'CRAZY_DIAMOND_PULSE',
        energy: 10,
        lastInteractionTimestamp: DateTime(2026, 10, 31),
      );

      final result = GenericUserActionExecutor.execute(
        currentKaiju: exhaustedKaiju,
        config: playConfig,
        triggeredAt: DateTime(2026, 10, 31),
      );

      expect(result.success, isFalse);
      expect(
        result.message,
        '> [ALERT] CRAZY_DIAMOND_PULSE is too exhausted to play! Needs dormancy.',
      );
    });

    test('GenericCustomizationRenderer resolves dynamic palette tokens', () {
      final palettes = PluginRegistry()
          .getCustomizationsByCategory(CustomizationCategory.palette);
      final amber = GenericCustomizationRenderer.resolvePalette(
        activePaletteId: 'palette_amber_monitor',
        palettePlugins: palettes,
      );

      expect(amber.id, 'palette_amber_monitor');
      expect(amber.foreground.value, 0xFFFFB000);
    });
  });
}
```

```dart
import 'package:flutter_test/flutter_test.dart';
import 'package:tamagotchi_mobile/models/age_phase.dart';
import 'package:tamagotchi_mobile/plugins/plugin_registry.dart';
import 'package:tamagotchi_mobile/state/kaiju_state_notifier.dart';
import 'package:tamagotchi_mobile/state/offline_storage_service.dart';

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();

  group('KaijuStateNotifier', () {
    late OfflineStorageService storage;
    late KaijuStateNotifier notifier;

    setUp(() async {
      await PluginRegistry().loadBundledPocManifest();
      storage = OfflineStorageService();
      notifier = KaijuStateNotifier(storage: storage);
    });

    tearDown(() {
      notifier.dispose();
    });

    test('starts with 240 credits and allows buying 2 Kaijus at 120 CR each', () {
      expect(notifier.state.credits, 240);
      expect(notifier.state.ownedKaijus, isEmpty);

      final bought1 = notifier.buyKaiju(kaijuType: 'godzilla', price: 120);
      expect(bought1, isTrue);
      expect(notifier.state.credits, 120);
      expect(notifier.state.ownedKaijus.length, 1);
      expect(notifier.state.activeKaijuId, notifier.state.ownedKaijus.first.id);

      final bought2 = notifier.buyKaiju(kaijuType: 'cyber_godzilla', price: 120);
      expect(bought2, isTrue);
      expect(notifier.state.credits, 0);
      expect(notifier.state.ownedKaijus.length, 2);

      // 3rd purchase fails due to 0 credits
      final bought3 = notifier.buyKaiju(kaijuType: 'godzilla', price: 120);
      expect(bought3, isFalse);
      expect(notifier.state.ownedKaijus.length, 2);
    });

    test('switches active Kaiju and executes eat action clamping hunger at 100', () {
      notifier.buyKaiju(kaijuType: 'godzilla', price: 120);
      notifier.buyKaiju(kaijuType: 'cyber_godzilla', price: 120);

      final secondId = notifier.state.ownedKaijus[1].id;
      notifier.switchActiveKaiju(secondId);
      expect(notifier.state.activeKaijuId, secondId);

      final eatConfig = PluginRegistry()
          .getEnabledUserActions(kaijuPhase: AgePhase.child)
          .firstWhere((a) => a.id == 'eat');

      final res = notifier.executePluginAction(
        eatConfig,
        autoResetToIdle: false,
      );
      expect(res?.success, isTrue);

      final updated = notifier.state.activeKaiju!;
      expect(updated.hunger, 100); // 80 + 25 clamped to 100
      expect(updated.energy, 75); // 80 - 5
      expect(updated.happiness, 85); // 80 + 5
      expect(updated.currentActionState, 'eating');
    });
  });
}
```

```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:tamagotchi_mobile/app.dart';
import 'package:tamagotchi_mobile/plugins/plugin_registry.dart';
import 'package:tamagotchi_mobile/state/offline_storage_service.dart';
import 'package:tamagotchi_mobile/state/settings_notifier.dart';

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('Full PoC end-to-end user flow: Shop -> Room -> Actions -> Swap -> Conf -> EvoMgr',
      (tester) async {
    await PluginRegistry().loadBundledPocManifest();
    final storage = OfflineStorageService();

    await tester.pumpWidget(
      ProviderScope(
        overrides: [
          offlineStorageProvider.overrideWithValue(storage),
        ],
        child: const UnixTamagotchiApp(useGoogleFontsNetwork: false),
      ),
    );
    await tester.pump();

    // 1. Starts on 02:SHOP with 240 CR because no Kaijus are owned yet
    expect(find.text('CREDITS: 240 CR'), findsOneWidget);
    expect(find.text('[ > ACQUIRE ]'), findsOneWidget);

    // 2. Acquire Godzilla (120 CR) -> auto-navigates to 01:ROOM
    await tester.tap(find.text('[ > ACQUIRE ]'));
    await tester.pump();

    expect(find.text('CR: 120'), findsOneWidget);
    expect(find.text('VITALS.ENERGY'), findsOneWidget);
    expect(find.text('VITALS.SATUR'), findsOneWidget);
    expect(find.text('VITALS.JOY'), findsOneWidget);

    // 3. Execute [ > EAT ] action from dynamic plugin bar
    expect(find.text('[ > EAT ]'), findsOneWidget);
    expect(find.text('[ > SLEEP ]'), findsOneWidget);
    expect(find.text('[ > PLAY ]'), findsOneWidget);

    await tester.tap(find.text('[ > EAT ]'));
    await tester.pump();

    // Hunger 80 + 25 -> 100%, Energy 80 - 5 -> 75%, Joy 80 + 5 -> 85%
    expect(find.text('100%'), findsOneWidget);
    expect(find.text('75%'), findsOneWidget);
    expect(find.text('85%'), findsOneWidget);

    // Let action duration timer elapse cleanly
    await tester.pump(const Duration(seconds: 4));

    // 4. Navigate to [03:SWAP] and verify active specimen card
    await tester.tap(find.text('[03:SWAP]'));
    await tester.pump();
    expect(find.text('* ACTIVE SPECIMEN IN HABITAT'), findsOneWidget);

    // 5. Navigate to [04:CONF], switch to CUSTOMIZE sub-tab, select AMBER_MONITOR
    await tester.tap(find.text('[04:CONF]'));
    await tester.pump();
    expect(find.text('SYS_DIAG'), findsOneWidget);
    expect(find.text('CUSTOMIZE'), findsOneWidget);

    await tester.tap(find.text('CUSTOMIZE'));
    await tester.pump();
    expect(find.textContaining('AMBER_MONITOR'), findsOneWidget);
    await tester.tap(find.textContaining('AMBER_MONITOR'));
    await tester.pump();

    // 6. Switch back to SYS_DIAG and open EVO_MGR teaser
    await tester.tap(find.text('SYS_DIAG'));
    await tester.pump();
    await tester.tap(find.text('[ > EVO_MGR // GROWTH_TIMELINE ]'));
    await tester.pump();

    expect(find.text('HISTORICAL_EVOLUTION'), findsOneWidget);
    await tester.tap(find.text('[ RETURN ]'));
    await tester.pump();
  });
```
